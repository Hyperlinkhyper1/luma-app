 function animate(now) {
      requestAnimationFrame(animate);
      const elapsed = now - introStartTime;
      const dt = Math.min(.05, Math.max(.001, (now - (animate.last || now)) / 1000));
      animate.last = now;
      if (!introFinished) {
        const progress = THREE.MathUtils.clamp(elapsed / 2500, 0, 1);
        const eased = 1 - Math.pow(1 - progress, 3);
        camera.position.lerpVectors(introStartPosition, introEndPosition, eased);
        camera.lookAt(lookStart.clone().lerp(lookEnd, eased));
        if (progress >= 1) {
          introFinished = true;
          controls.target.copy(lookEnd);
          lastActivity = now;
        }
      } else {
        controls.autoRotate = now - lastActivity > 4200;
        controls.update();
      }
      updateEnvironment(elapsed);
      updateDoors(dt);
      updatePetals(now, dt);
      updateDetails(now);
      renderer.render(scene, camera);
      frames++;
      if (now - fpsClock > 1000) {
        const fps = Math.round(frames * 1000 / (now - fpsClock));
        window.__VOXEL__.fps = fps;
        stats.textContent = `${Math.round(voxelCount / 1000)}k voxels | ${fps} FPS`;
        frames = 0;
        fpsClock = now;
      }
    }
